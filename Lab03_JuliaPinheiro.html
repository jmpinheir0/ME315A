getStats <- function(input, pos) {
  
  input %>%
    filter(!grepl("^1", DESTINATION_AIRPORT)) %>%
    filter(
      !is.na(DESTINATION_AIRPORT),
      !is.na(ARRIVAL_DELAY)
    ) %>%
    group_by(DESTINATION_AIRPORT) %>%
    summarise(
      soma = sum(ARRIVAL_DELAY),
      total = n(),
      .groups = "drop"
    )
}

computeStats <- function(stats) {
  
  stats %>%
    group_by(DESTINATION_AIRPORT) %>%
    summarise(
      soma = sum(soma),
      total = sum(total),
      .groups = "drop"
    ) %>%
    mutate(
      media = soma / total
    )
}

flights <- computeStats(flights)
